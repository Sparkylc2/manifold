# so editing a demo rebuilds only that demo and the 37 build in parallel.
# a name here is both factories/<name>.cpp and Demo::make_<name>()

# editing a shared header (renderer/, fluid/, solver/) fans out to every demo
# TU, so iterating on one demo means paying for all 37; this cuts the set
set(MANIFOLD_DEMOS "all" CACHE STRING
    "Demos to compile: 'all', or a ;-separated subset of MANIFOLD_ALL_DEMOS")

if(MANIFOLD_DEMOS STREQUAL "all")
    set(_enabled ${MANIFOLD_ALL_DEMOS})
else()
    set(_enabled ${MANIFOLD_DEMOS})
    foreach(d ${_enabled})
        if(NOT d IN_LIST MANIFOLD_ALL_DEMOS)
            message(FATAL_ERROR
                "MANIFOLD_DEMOS: unknown demo '${d}'.\n"
                "Known demos: ${MANIFOLD_ALL_DEMOS}")
        endif()
    endforeach()
endif()

set(_disabled ${MANIFOLD_ALL_DEMOS})
if(_enabled)
    list(REMOVE_ITEM _disabled ${_enabled})
endif()

set(_demo_srcs "")
foreach(d ${_enabled})
    list(APPEND _demo_srcs factories/${d}.cpp)
endforeach()

# the launcher registry names every factory unconditionally, so demos left out
# of the build still have to resolve; they get a stub returning nullptr, and
# demo_stubbed lets the launcher tell those apart without constructing anything
set(_stub "// generated by demos/CMakeLists.txt -- do not edit\n\n")
string(APPEND _stub "#include \"${CMAKE_CURRENT_SOURCE_DIR}/demo_factories.h\"\n\n")
string(APPEND _stub "#include <algorithm>\n#include <vector>\n\n")
string(APPEND _stub "namespace manifold::Demo {\n\n")
foreach(d ${_disabled})
    string(APPEND _stub
        "std::unique_ptr<DemoBase> make_${d}() { return nullptr; }\n")
endforeach()
string(APPEND _stub "\nbool demo_stubbed(DemoFactory factory) {\n")
string(APPEND _stub "    static const std::vector<DemoFactory> stubs = {\n")
foreach(d ${_disabled})
    string(APPEND _stub "        &make_${d},\n")
endforeach()
string(APPEND _stub "    };\n")
string(APPEND _stub
    "    return std::find(stubs.begin(), stubs.end(), factory) != stubs.end();\n")
string(APPEND _stub "}\n\n} // namespace manifold::Demo\n")

set(_stub_cpp ${CMAKE_CURRENT_BINARY_DIR}/disabled_demos.cpp)
file(GENERATE OUTPUT ${_stub_cpp} CONTENT "${_stub}")
set_source_files_properties(${_stub_cpp} PROPERTIES GENERATED TRUE)

list(LENGTH _enabled _n_enabled)
list(LENGTH MANIFOLD_ALL_DEMOS _n_all)
if(_disabled)
    message(STATUS
        "manifold demos: ${_n_enabled}/${_n_all} compiled (${_enabled}); "
        "the rest are stubbed out")
else()
    message(STATUS "manifold demos: all ${_n_all} compiled")
endif()

add_library(manifold_demos STATIC ${_demo_srcs} ${_stub_cpp})

target_include_directories(manifold_demos PUBLIC
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/include
)
target_link_libraries(manifold_demos PUBLIC
    manifold_solver
    manifold_control
    manifold_pde
    manifold_ai
    manifold_fea
    manifold_fluid
    manifold_electrical
    manifold_renderer
    raygui
    Threads::Threads
)

add_subdirectory(launcher)
